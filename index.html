    <script>

        let allGamesData = [];


        async function fetchScores() {

            const container = document.getElementById('games-container');

            const statusEl = document.getElementById('update-status');

            statusEl.textContent = 'Updating...';


            try {

                // ESPN public CFB scoreboard endpoint

                const response = await fetch('https://site.api.espn.com/apis/site/v2/sports/football/college-football/scoreboard');

                if (!response.ok) throw new Error('Network response was not ok');

                

                const data = await response.json();

                allGamesData = data.events || [];

                

                renderGames();

                statusEl.textContent = 'Live';

            } catch (error) {

                console.error("Error fetching scores:", error);

                statusEl.textContent = 'Error updating';

                if (allGamesData.length === 0) {

                    container.innerHTML = `<div class="error">Unable to load live scores right now. Please check your connection or try refreshing.</div>`;

                }

            }

        }


        function filterGames() {

            renderGames();

        }


        function renderGames() {

            const container = document.getElementById('games-container');

            const filter = document.getElementById('filter-select').value;


            if (!allGamesData || allGamesData.length === 0) {

                container.innerHTML = `<div class="no-games">No games scheduled or available at the moment.</div>`;

                return;

            }


            let filtered = allGamesData.filter(event => {

                const status = event.status.type.state; // 'pre', 'in', 'post'

                const competitions = event.competitions[0];

                const homeTeam = competitions.competitors.find(c => c.homeAway === 'home');

                const awayTeam = competitions.competitors.find(c => c.homeAway === 'away');

                const isTop25 = (homeTeam.curatedRank?.current <= 25) || (awayTeam.curatedRank?.current <= 25);


                if (filter === 'live') return status === 'in';

                if (filter === 'final') return status === 'post';

                if (filter === 'top25') return isTop25;

                return true; // 'all'

            });


            // If user selected 'Live Only' but there are no live games, fallback or show message

            if (filtered.length === 0) {

                container.innerHTML = `<div class="no-games">No games match the selected filter ("${filter}"). Try switching to "All Games".</div>`;

                return;

            }


            container.innerHTML = filtered.map(event => {

                const comp = event.competitions[0];

                const statusState = event.status.type.state; // pre, in, post

                const statusDetail = event.status.type.detail; // e.g., "3rd - 4:12", "Final", "Sat, 7:00 PM"

                const broadcast = comp.broadcasts?.[0]?.names?.[0] || '';

                const situation = comp.situation ? comp.situation.downDistanceText : '';


                const homeTeam = comp.competitors.find(c => c.homeAway === 'home');

                const awayTeam = comp.competitors.find(c => c.homeAway === 'away');


                const homeScore = parseInt(homeTeam.score || 0);

                const awayScore = parseInt(awayTeam.score || 0);


                let homeWinner = false;

                let awayWinner = false;

                if (statusState === 'post') {

                    if (homeScore > awayScore) homeWinner = true;

                    else if (awayScore > homeScore) awayWinner = true;

                }


                // Format status badge class

                let statusClass = 'upcoming';

                let statusText = statusDetail;

                if (statusState === 'in') {

                    statusClass = 'live';

                } else if (statusState === 'post') {

                    statusClass = 'final';

                    statusText = 'FINAL';

                }


                const homeRank = homeTeam.curatedRank?.current <= 25 ? `#${homeTeam.curatedRank.current} ` : '';

                const awayRank = awayTeam.curatedRank?.current <= 25 ? `#${awayTeam.curatedRank.current} ` : '';


                return `

                    <div class="game-card">

                        <div class="game-header">

                            <span class="game-status ${statusClass}">${statusText}</span>

                            <span class="broadcast">${broadcast}</span>

                        </div>

                        <div class="teams-list">

                            <div class="team-row ${awayWinner ? 'winner' : ''}">

                                <div class="team-info">

                                    <img class="team-logo" src="${awayTeam.team.logo || ''}" alt="" onerror="this.style.display='none'">

                                    <span class="team-name">${awayRank}${awayTeam.team.shortDisplayName || awayTeam.team.name}</span>

                                </div>

                                <span class="team-score">${awayTeam.score ?? '0'}</span>

                            </div>

                            <div class="team-row ${homeWinner ? 'winner' : ''}">

                                <div class="team-info">

                                    <img class="team-logo" src="${homeTeam.team.logo || ''}" alt="" onerror="this.style.display='none'">

                                    <span class="team-name">${homeRank}${homeTeam.team.shortDisplayName || homeTeam.team.name}</span>

                                </div>

                                <span class="team-score">${homeTeam.score ?? '0'}</span>

                            </div>

                        </div>

                        ${situation || (statusState === 'in' && event.status.period) ? `

                        <div class="game-footer">

                            <span class="situation">${situation || ''}</span>

                            <span></span>

                        </div>` : ''}

                    </div>

                `;

            }).join('');

        }


        // Initial fetch

        fetchScores();


        // Auto-refresh every 20 seconds for live monitoring

        setInterval(fetchScores, 20000);

    </script>
